cmake_minimum_required(VERSION 3.24)
project(NeonBorough VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NB_BUILD_GAME "Build the raylib desktop game" ON)
option(NB_BUILD_TESTS "Build deterministic headless tests" ON)
option(NB_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

add_library(neon_core src/world.cpp src/simulation.cpp src/save.cpp)
target_include_directories(neon_core PUBLIC include)
if(MSVC)
    target_compile_options(neon_core PRIVATE /W4 /permissive- /utf-8)
    target_compile_definitions(neon_core PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(neon_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(NB_SANITIZERS AND NOT MSVC)
    target_compile_options(neon_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(neon_core PUBLIC -fsanitize=address,undefined)
endif()

if(NB_BUILD_GAME)
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
    # raylib 5.5, pinned to its immutable release commit.
    FetchContent_Declare(raylib
        URL https://github.com/raysan5/raylib/archive/c1ab645ca298a2801097931d1079b10ff7eb9df8.tar.gz
        DOWNLOAD_EXTRACT_TIMESTAMP FALSE)
    FetchContent_MakeAvailable(raylib)
    add_executable(neon_borough src/main.cpp src/presentation.cpp src/audio.cpp)
    target_link_libraries(neon_borough PRIVATE neon_core raylib)
    if(MSVC)
        target_compile_options(neon_borough PRIVATE /W4 /utf-8)
    else()
        target_compile_options(neon_borough PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(MINGW)
        target_link_options(neon_borough PRIVATE -static-libgcc -static-libstdc++)
    elseif(MSVC)
        target_link_options(neon_borough PRIVATE /SUBSYSTEM:CONSOLE)
    endif()
    install(TARGETS neon_borough RUNTIME DESTINATION .)
    install(FILES README.md LICENSE THIRD_PARTY.md CHANGELOG.md DESTINATION .)
    install(DIRECTORY docs/ DESTINATION docs)
    install(FILES "${raylib_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME raylib.txt)
    install(FILES "${raylib_SOURCE_DIR}/src/external/glfw/LICENSE.md" DESTINATION licenses RENAME glfw.txt)
endif()

if(NB_BUILD_TESTS)
    enable_testing()
    add_executable(neon_tests tests/core_tests.cpp)
    target_link_libraries(neon_tests PRIVATE neon_core)
    add_test(NAME neon_core_tests COMMAND neon_tests)
    set_tests_properties(neon_core_tests PROPERTIES TIMEOUT 180)
endif()
